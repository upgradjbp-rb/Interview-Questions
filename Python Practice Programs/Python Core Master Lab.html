Python 3 
def check_palindrome_math(num):
    temp = num
    reversed_num = 0
    while temp > 0:
        reversed_num = (reversed_num * 10) + (temp % 10)
        temp //= 10
    
    if num == reversed_num:
        print(f"Original number {num}\nYes. given number is palindrome number")
    else:
        print(f"Original number {num}\nNo. given number is not palindrome number")

check_palindrome_math(121)

class Car:
    def __init__(self, make, model, year):
        self.make = make
        self.model = model
        self.year = year

    def start_engine(self):
        print(f"The {self.year} {self.make} {self.model}'s engine is now running!")

# Instantiation & Method invocation
my_car = Car("Toyota", "Camry", 2022)
my_car.start_engine()
Key Concept: self binds instance variables to specific object instances, enabling clean object encapsulation.
